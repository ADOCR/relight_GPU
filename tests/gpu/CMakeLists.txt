cmake_minimum_required(VERSION 3.18)
project(relight_gpu_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(CUDAToolkit QUIET)
add_executable(projection_test projection_test.cpp ../../src/compute/gpu_projection.cpp)
target_link_libraries(projection_test PRIVATE Qt6::Core)
if(CUDAToolkit_FOUND)
    target_include_directories(projection_test PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
    target_compile_definitions(projection_test PRIVATE RELIGHT_HAS_CUDA=1)
else()
    target_compile_definitions(projection_test PRIVATE RELIGHT_HAS_CUDA=0)
endif()
if(MSVC)
    target_compile_options(projection_test PRIVATE /EHsc /utf-8)
endif()
add_executable(projection_cpu_stub_test cpu_stub_test.cpp ../../src/compute/gpu_projection.cpp ../../src/compute/gpu_fft.cpp)
target_link_libraries(projection_cpu_stub_test PRIVATE Qt6::Core)
target_compile_definitions(projection_cpu_stub_test PRIVATE RELIGHT_HAS_CUDA=0)
enable_testing()
add_test(NAME cuda_projection COMMAND projection_test)
add_test(NAME cpu_only_backend COMMAND projection_cpu_stub_test)
add_executable(channel_projection_test channel_projection_test.cpp)
add_test(NAME channel_projection_exact COMMAND channel_projection_test)
if(MSVC)
    target_compile_options(channel_projection_test PRIVATE /EHsc /fp:precise)
endif()
# Optional override for localized MSVC installations whose /showIncludes prefix
# is decoded incorrectly by CMake. Empty preserves CMake's normal detection.
set(RELIGHT_MSVC_DEPS_PREFIX "" CACHE STRING "MSVC /showIncludes prefix override")
if(MSVC AND RELIGHT_MSVC_DEPS_PREFIX)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${RELIGHT_MSVC_DEPS_PREFIX}")
endif()
add_executable(separated_gpu_test separated_gpu_test.cpp ../../src/compute/gpu_projection.cpp)
target_link_libraries(separated_gpu_test PRIVATE Qt6::Core)
if(CUDAToolkit_FOUND)
    target_include_directories(separated_gpu_test PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
    target_compile_definitions(separated_gpu_test PRIVATE RELIGHT_HAS_CUDA=1)
else()
    target_compile_definitions(separated_gpu_test PRIVATE RELIGHT_HAS_CUDA=0)
endif()
add_test(NAME separated_rgb_cuda COMMAND separated_gpu_test)
add_executable(fft_test fft_test.cpp ../../src/compute/gpu_fft.cpp)
target_link_libraries(fft_test PRIVATE Qt6::Core)
if(CUDAToolkit_FOUND)
    target_include_directories(fft_test PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
    target_compile_definitions(fft_test PRIVATE RELIGHT_HAS_CUDA=1)
else()
    target_compile_definitions(fft_test PRIVATE RELIGHT_HAS_CUDA=0)
endif()
add_test(NAME double_precision_fft_cuda COMMAND fft_test)
if(CUDAToolkit_FOUND AND TARGET CUDA::nvjpeg)
    add_executable(nvjpeg_benchmark nvjpeg_benchmark.cpp ../../src/jpeg_decoder.cpp)
    target_include_directories(nvjpeg_benchmark PRIVATE ../../src ../../external/libjpeg-turbo-2.0.6/include)
    target_link_libraries(nvjpeg_benchmark PRIVATE Qt6::Core CUDA::nvjpeg CUDA::cudart ${CMAKE_CURRENT_SOURCE_DIR}/../../external/libjpeg-turbo-2.0.6/lib/jpeg.lib)
endif()
add_executable(surface_test surface_test.cpp ../../src/normals/fft_normal_integration.cpp ../../src/compute/gpu_fft.cpp)
target_include_directories(surface_test PRIVATE ../../external/eigen-3.3.9)
target_link_libraries(surface_test PRIVATE Qt6::Core)
if(CUDAToolkit_FOUND)
    target_include_directories(surface_test PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
    target_compile_definitions(surface_test PRIVATE RELIGHT_HAS_CUDA=1)
else()
    target_compile_definitions(surface_test PRIVATE RELIGHT_HAS_CUDA=0)
endif()
add_test(NAME surface_cpu_cuda_consistency COMMAND surface_test 128)

set_tests_properties(cuda_projection separated_rgb_cuda double_precision_fft_cuda PROPERTIES SKIP_RETURN_CODE 77)

set_tests_properties(surface_cpu_cuda_consistency PROPERTIES SKIP_RETURN_CODE 77)
